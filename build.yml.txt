name: Build APK
on:
  push:
    branches: [main]
  workflow_dispatch:
jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-node@v4
        with: {node-version: 20}
      - uses: actions/setup-java@v4
        with: {distribution: temurin, java-version: 17}
      - run: |
          npm init -y
          npm i @capacitor/core@6 @capacitor/android@6 @capacitor/cli@6 three@0.160.0
          mkdir -p www && cp index.html www/ && cp node_modules/three/build/three.module.js www/
          npx cap init "City Transit Tycoon" com.citytransit.tycoon --web-dir www
          npx cap add android
          npx cap sync android
          cd android && chmod +x gradlew && ./gradlew assembleDebug
      - uses: actions/upload-artifact@v4
        with:
          name: CityTransitTycoon3D-v2-apk
          path: android/app/build/outputs/apk/debug/app-debug.apk
